console.log("AI Agent szkript betöltve.");

chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
  if (request.action === "run_command") {
    console.log("Utasítás érkezett:", request.text);
    
    // Itt történik a varázslat: Megpróbáljuk megtalálni az elemet a szöveg alapján.
    // Egy igazi AI Agent itt kapna egy pontos CSS SELECTOR-t az AI-tól.
    // Most egy egyszerű kereső algoritmust használunk (Heurisztika).
    
    const result = executeHeuristicAction(request.text);
    sendResponse({ status: result });
  }
});

function executeHeuristicAction(command) {
  const lowerCommand = command.toLowerCase();

  // 1. KATTINTÁS (Click) detektálása
  if (lowerCommand.includes("kattints") || lowerCommand.includes("nyomd meg")) {
    // Kinyerjük a kulcsszót (pl. "Kattints a belépés gombra" -> keresett szó: "belépés")
    // Ez egy nagyon egyszerűsített logika!
    const keywords = lowerCommand.replace("kattints", "").replace("a", "").replace("gombra", "").replace("linkre", "").trim();
    
    return findAndClickElement(keywords);
  }

  // 2. GÉPELÉS (Input) detektálása
  if (lowerCommand.includes("írd be")) {
    // Pl: "Írd be a keresőbe hogy alma"
    // Ez bonyolultabb, itt most csak példa szinten kezeljük.
    return "Gépelés funkció még fejlesztés alatt (AI API szükséges).";
  }

  return "Nem értettem a parancsot.";
}

function findAndClickElement(searchText) {
  // Megkeresünk minden gombot és linket az oldalon
  const elements = document.querySelectorAll('button, a, input[type="submit"], [role="button"]');
  
  for (let el of elements) {
    // Megnézzük, hogy az elem szövege vagy ID-ja tartalmazza-e a keresett szót
    const text = (el.innerText || el.value || el.getAttribute('aria-label') || "").toLowerCase();
    
    if (text.includes(searchText.toLowerCase())) {
      console.log("Elem megtalálva:", el);
      
      // Kiemeljük az elemet pirossal, hogy lásd, mit talált meg
      el.style.border = "2px solid red";
      
      // KATTINTÁS
      el.click();
      return `Rákattintottam erre: "${text}"`;
    }
  }
  return "Nem találtam ilyen elemet.";
}
