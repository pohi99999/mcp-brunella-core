document.getElementById('execute').addEventListener('click', async () => {
  const command = document.getElementById('command').value;
  const statusDiv = document.getElementById('status');
  
  if (!command) return;

  statusDiv.textContent = "Feldolgozás...";

  // 1. Lépés: Itt hívnád meg az AI-t (OpenAI/Gemini API).
  // Most "szimuláljuk" az AI-t: egyszerűen továbbítjuk a parancsot a Content Scriptnek,
  // ami megpróbálja kitalálni, melyik elemre gondoltál.
  
  // Lekérjük az aktív tabot
  const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });

  // Üzenet küldése a weboldalnak (content.js)
  chrome.tabs.sendMessage(tab.id, { action: "run_command", text: command }, (response) => {
    if (chrome.runtime.lastError) {
      statusDiv.textContent = "Hiba: Frissítsd az oldalt!";
    } else {
      statusDiv.textContent = response ? response.status : "Kész.";
    }
  });
});
