python# Custom Python Component
import re
import uuid
from datetime import datetime

def parse_markdown_to_structured(markdown_text: str) -> list[dict]:
    """Markdown → Structured list konverzió"""
    entries = []
    current_category = None
    
    for line in markdown_text.split('\n'):
        if line.startswith('## '):
            current_category = line.replace('## ', '').strip()
        elif line.startswith('- **'):
            # Regex: **Title** (URL)
            match = re.search(r'\*\*(.+?)\*\*.*?\((.+?)\)', line)
            if match:
                entries.append({
                    'id': str(uuid.uuid4()),
                    'title': match.group(1),
                    'url': match.group(2),
                    'category': current_category,
                    'raw_text': line
                })
    return entries

python# Custom Component - Parallel mentés
import json
from datetime import datetime

def save_to_knowledge_base(json_data: list[dict]):
    # 1. ChromaDB vectorstore (embeddings generálás)
    from chromadb import Client
    client = Client()
    collection = client.get_or_create_collection("ai_innovations")
    
    for entry in json_data:
        collection.add(
            ids=[entry['id']],
            documents=[f"{entry['title']}: {entry['summary']}"],
            metadatas=[{
                'url': entry['url'],
                'category': entry['category'],
                'date': entry['discovered_at']
            }]
        )
    
    # 2. Google Drive JSON backup
    filename = f"ai_innovations_{datetime.now().strftime('%Y%m%d')}.json"
    # Drive API upload (később implementálható)
    with open(f'/tmp/{filename}', 'w', encoding='utf-8') as f:
        json.dump(json_data, f, ensure_ascii=False, indent=2)
    
    return {
        'chromadb_count': len(json_data),
        'backup_file': filename
    }

pythondef remove_duplicates(entries: list[dict]) -> list[dict]:
    seen_urls = set()
    unique = []
    for e in entries:
        if e['url'] not in seen_urls:
            unique.append(e)
            seen_urls.add(e['url'])
    return unique
