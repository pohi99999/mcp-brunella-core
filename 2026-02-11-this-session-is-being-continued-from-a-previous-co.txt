  import { useState, useEffect, useRef } from 'react';
  import { Card, CardContent, CardHeader, CardTitle } from
  '@/components/ui/card';
  import { Button } from '@/components/ui/button';
  import { Input } from '@/components/ui/input';
  import { ScrollArea } from '@/components/ui/scroll-area';
  import { Send, RefreshCw } from 'lucide-react';
  import { logInfo, logError } from '@/utils/logger';

  interface Message {
    id: string;
    type: 'user' | 'agent' | 'system';
    from: string;
    content: string;
    timestamp: number;
  }

  export function CloudflareChat() {
    const [messages, setMessages] = useState<Message[]>([]);
    const [input, setInput] = useState('');
    const [connected, setConnected] = useState(false);
    const [loading, setLoading] = useState(false);
    const wsRef = useRef<WebSocket | null>(null);
    const scrollRef = useRef<HTMLDivElement>(null);

    // WebSocket Connection
    useEffect(() => {
      connectWebSocket();
      return () => {
        if (wsRef.current) {
          wsRef.current.close();
        }
      };
    }, []);

    const connectWebSocket = () => {
      try {
        // Cloudflare Workers WebSocket endpoint
        const ws = new WebSocket('wss://chat-bas.peterpohanka.com/ws');

        ws.onopen = () => {
          setConnected(true);
          logInfo('CloudflareChat', 'WebSocket connected');
          addSystemMessage('Connected to Cloudflare Chat');
        };

        ws.onmessage = (event) => {
          try {
            const data = JSON.parse(event.data);
            handleIncomingMessage(data);
          } catch (e) {
            logError('CloudflareChat', 'Failed to parse message', e);
          }
        };

        ws.onerror = (error) => {
          logError('CloudflareChat', 'WebSocket error', error);
          setConnected(false);
        };

        ws.onclose = () => {
          setConnected(false);
          logInfo('CloudflareChat', 'WebSocket disconnected');
          addSystemMessage('Disconnected. Retrying in 5s...');
          setTimeout(connectWebSocket, 5000);
        };

        wsRef.current = ws;
      } catch (e) {
        logError('CloudflareChat', 'Failed to connect', e);
      }
    };

    const handleIncomingMessage = (data: any) => {
      const message: Message = {
        id: crypto.randomUUID(),
        type: data.type || 'agent',
        from: data.from || 'Agent',
        content: data.content,
        timestamp: data.timestamp || Date.now(),
      };
      setMessages((prev) => [...prev, message]);
      scrollToBottom();
    };

    const addSystemMessage = (content: string) => {
      setMessages((prev) => [
        ...prev,
        {
          id: crypto.randomUUID(),
          type: 'system',
          from: 'System',
          content,
          timestamp: Date.now(),
        },
      ]);
    };

    const sendMessage = async () => {
      if (!input.trim() || !wsRef.current || wsRef.current.readyState !==
  WebSocket.OPEN) {
        return;
      }

      const userMessage: Message = {
        id: crypto.randomUUID(),
        type: 'user',
        from: 'You',
        content: input,
        timestamp: Date.now(),
      };

      setMessages((prev) => [...prev, userMessage]);
      setLoading(true);

      try {
        // Send to WebSocket
        wsRef.current.send(
          JSON.stringify({
            type: 'chat',
            from: 'dashboard-user',
            content: input,
            timestamp: Date.now(),
          })
        );

        // Also send to local backend for agent processing
        const response = await
  fetch('http://localhost:3000/api/agents/orchestrator/execute', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ task: input }),
        });

        if (response.ok) {
          const result = await response.json();
          logInfo('CloudflareChat', 'Backend response received', result);
        }
      } catch (e) {
        logError('CloudflareChat', 'Failed to send message', e);
        addSystemMessage('Error sending message. Please try again.');
      } finally {
        setInput('');
        setLoading(false);
        scrollToBottom();
      }
    };

    const scrollToBottom = () => {
      setTimeout(() => {
        if (scrollRef.current) {
          scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
        }
      }, 100);
    };

    return (
      <Card className="h-full flex flex-col">
        <CardHeader className="flex-row items-center justify-between">
          <CardTitle>Cloudflare Chat</CardTitle>
          <div className="flex items-center gap-2">
            <div className={`w-2 h-2 rounded-full ${connected ? 'bg-green-500' :
   'bg-red-500'}`} />
            <span className="text-sm text-muted-foreground">
              {connected ? 'Connected' : 'Disconnected'}
            </span>
            <Button variant="ghost" size="icon" onClick={connectWebSocket}>
              <RefreshCw className="h-4 w-4" />
            </Button>
          </div>
        </CardHeader>
      
        <CardContent className="flex-1 flex flex-col p-0">
          <ScrollArea ref={scrollRef} className="flex-1 p-4">
            {messages.map((msg) => (
              <div
                key={msg.id}
                className={`mb-4 ${
                  msg.type === 'user' ? 'text-right' : msg.type === 'system' ? 
  'text-center' : 'text-left'
                }`}
              >
                <div
                  className={`inline-block max-w-[80%] p-3 rounded-lg ${
                    msg.type === 'user'
                      ? 'bg-primary text-primary-foreground'
                      : msg.type === 'system'
                      ? 'bg-muted text-muted-foreground text-sm'
                      : 'bg-secondary'
                  }`}
                >
                  {msg.type !== 'system' && (
                    <div className="text-xs opacity-70 mb-1">{msg.from}</div>
                  )}
                  <div>{msg.content}</div>
                  <div className="text-xs opacity-50 mt-1">
                    {new Date(msg.timestamp).toLocaleTimeString()}
                  </div>
                </div>
              </div>
            ))}
          </ScrollArea>

          <div className="p-4 border-t flex gap-2">
            <Input
              value={input}
              onChange={(e) => setInput(e.target.value)}
              onKeyPress={(e) => e.key === 'Enter' && !e.shiftKey &&
  sendMessage()}
              placeholder="Type a message..."
              disabled={!connected || loading}
            />
            <Button onClick={sendMessage} disabled={!connected || loading || 
  !input.trim()}>
              <Send className="h-4 w-4" />
            </Button>
          </div>
        </CardContent>
      </Card>
    );
  }

  // src/agents/task-decomposer.ts
  import { IAgent, AgentResponse } from './types.js';
  import { logInfo, setAgentStatus } from '../utils/logger.js';
  import { AgentManager } from './manager.js';

  interface MicroTask {
    id: string;
    agent: string;
    task: string;
    dependencies: string[];  // Task IDs that must complete first
    parallel: boolean;        // Can run in parallel with others
    retries: number;
    timeout: number;          // ms
  }

  export class TaskDecomposerAgent implements IAgent {
    name = "TaskDecomposer";
    role = "Complex Task Orchestrator";
    description = "Breaks down complex tasks into micro-tasks and orchestrates 
  micro-agents";
    capabilities = ["task_analysis", "dependency_resolution", 
  "parallel_execution", "fault_tolerance"];

    private agentManager: AgentManager;

    constructor() {
      this.agentManager = new AgentManager();
    }

    async execute(task: string, context?: any): Promise<AgentResponse> {
      setAgentStatus(this.name, 'working', 'Decomposing task...');

      try {
        // 1. Analyze task complexity
        const microTasks = await this.decomposeTask(task);
        logInfo(this.name, `Decomposed into ${microTasks.length} micro-tasks`);

        // 2. Build execution graph
        const executionPlan = this.buildExecutionGraph(microTasks);

        // 3. Execute micro-tasks
        const results = await this.executeMicroTasks(executionPlan);

        // 4. Aggregate results
        const finalResult = this.aggregateResults(results);

        return {
          status: "success",
          agent: this.name,
          result: finalResult,
          metadata: {
            micro_tasks: microTasks.length,
            execution_time: results.totalTime,
            parallel_tasks: results.parallelCount,
          }
        };
      } catch (e: any) {
        return { status: "error", agent: this.name, error: e.message };
      } finally {
        setAgentStatus(this.name, 'idle');
      }
    }

    private async decomposeTask(task: string): Promise<MicroTask[]> {
      // Use LLM to analyze task and break it down
      const prompt = `
  Analyze this task and break it into micro-tasks:
  "${task}"

  Return JSON array with structure:
  [
    {
      "id": "unique_id",
      "agent": "agent_name",
      "task": "specific subtask",
      "dependencies": ["task_id1", "task_id2"],
      "parallel": true/false,
      "retries": 3,
      "timeout": 30000
    }
  ]

  Available micro-agents:
  - FileReader: Read file contents
  - DataValidator: Validate data structure
  - CodeAnalyzer: Analyze code patterns
  - TestRunner: Run specific tests
  - DocumentWriter: Write documentation
  - APICaller: Make API requests
  - DataTransformer: Transform data format
  `;

      const response = await fetch('http://localhost:11434/api/generate', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: 'qwen2.5-coder:latest',
          prompt,
          stream: false,
        }),
      });

      const data = await response.json();
      const microTasks = JSON.parse(data.response);
      return microTasks;
    }

    private buildExecutionGraph(tasks: MicroTask[]): Map<string, MicroTask[]> {
      const graph = new Map<string, MicroTask[]>();
      const levels = new Map<string, number>();

      // Calculate dependency levels (0 = no deps, 1 = depends on level 0, etc.)
      const calculateLevel = (task: MicroTask): number => {
        if (levels.has(task.id)) return levels.get(task.id)!;
        
        if (task.dependencies.length === 0) {
          levels.set(task.id, 0);
          return 0;
        }

        const depLevels = task.dependencies.map(depId => {
          const depTask = tasks.find(t => t.id === depId);
          return depTask ? calculateLevel(depTask) : 0;
        });

        const level = Math.max(...depLevels) + 1;
        levels.set(task.id, level);
        return level;
      };

      tasks.forEach(task => calculateLevel(task));

      // Group tasks by level
      tasks.forEach(task => {
        const level = levels.get(task.id)!;
        const levelKey = `level_${level}`;
        if (!graph.has(levelKey)) {
          graph.set(levelKey, []);
        }
        graph.get(levelKey)!.push(task);
      });

      return graph;
    }

    private async executeMicroTasks(executionPlan: Map<string, MicroTask[]>):
  Promise<any> {
      const results = new Map<string, any>();
      let parallelCount = 0;
      const startTime = Date.now();

      // Execute level by level
      for (const [level, tasks] of Array.from(executionPlan.entries()).sort()) {
        logInfo(this.name, `Executing ${level}: ${tasks.length} tasks`);

        // Separate parallel and sequential tasks
        const parallelTasks = tasks.filter(t => t.parallel);
        const sequentialTasks = tasks.filter(t => !t.parallel);

        // Execute parallel tasks
        if (parallelTasks.length > 0) {
          parallelCount += parallelTasks.length;
          const promises = parallelTasks.map(task =>
  this.executeSingleTask(task));
          const parallelResults = await Promise.allSettled(promises);
          parallelResults.forEach((result, idx) => {
            results.set(parallelTasks[idx].id, result);
          });
        }

        // Execute sequential tasks
        for (const task of sequentialTasks) {
          const result = await this.executeSingleTask(task);
          results.set(task.id, result);
        }
      }

      return {
        results: Object.fromEntries(results),
        totalTime: Date.now() - startTime,
        parallelCount,
      };
    }

    private async executeSingleTask(task: MicroTask): Promise<any> {
      logInfo(this.name, `Executing micro-task: ${task.id} (${task.agent})`);

      let lastError: any;
      for (let attempt = 0; attempt < task.retries; attempt++) {
        try {
          const timeoutPromise = new Promise((_, reject) =>
            setTimeout(() => reject(new Error('Task timeout')), task.timeout)
          );

          const taskPromise = this.agentManager.executeAgent(task.agent,
  task.task);
          const result = await Promise.race([taskPromise, timeoutPromise]);

          return { success: true, result, attempt };
        } catch (e: any) {
          lastError = e;
          logInfo(this.name, `Retry ${attempt + 1}/${task.retries} for
  ${task.id}`);
          await new Promise(resolve => setTimeout(resolve, 1000 * (attempt + 
  1))); // Exponential backoff
        }
      }

      return { success: false, error: lastError.message };
    }

    private aggregateResults(execution: any): any {
      const successful = Object.values(execution.results).filter((r: any) =>
  r.success).length;
      const failed = Object.values(execution.results).filter((r: any) => 
  !r.success).length;

      return {
        summary: `${successful} successful, ${failed} failed`,
        details: execution.results,
        metrics: {
          total_time: execution.totalTime,
          parallel_tasks: execution.parallelCount,
        },
      };
    }
  }

      .command('status <name>')

  'track.md');

        try {
          const content = await fs.readFile(trackPath, 'utf-8');

          // Parse checkboxes
          const todos = content.match(/- \[(x| )\] .+/g) || [];
          const completed = todos.filter(t => t.includes('[x]')).length;
          const total = todos.length;
          const progress = Math.round((completed / total) * 100);

          // Parse status
          const statusMatch = content.match(/\*\*Status:\*\* (\w+)/);
          const status = statusMatch ? statusMatch[1] : 'UNKNOWN';

          console.log(chalk.bold(`\n📋 Track: ${name}`));

  ${getStatusColor(status)}${status}${chalk.reset()}`));

          console.log(chalk.bold('TODO List:'));
          todos.forEach((todo, i) => {

            const text = todo.replace(/- \[(x| )\] /, '');
            console.log(
              `${i + 1}. ${isDone ? chalk.green('✓') : chalk.gray('○')} ${
                isDone ? chalk.gray(text) : text
              }`
            );
          });

          // Test status
          console.log(chalk.bold('\n🧪 Test Status:'));
          const { execSync } = await import('child_process');
          try {
            execSync('npm test', { stdio: 'ignore' });
            console.log(chalk.green('✓ All tests passing'));
          } catch (e) {
            console.log(chalk.red('✗ Tests failing - must fix before
  completion'));
          }

        } catch (e: any) {
          console.error(chalk.red(`Error: Track not found at ${trackPath}`));
        }
      });

    tracks
      .command('list')
      .description('List all tracks')
      .action(async () => {
        const tracksDir = path.join(process.cwd(), 'conductor', 'tracks');
        const dirs = await fs.readdir(tracksDir);

        console.log(chalk.bold('\n📚 All Tracks:\n'));

          const trackPath = path.join(tracksDir, dir, 'track.md');
          try {

            console.log(`${getStatusIcon(status)} ${dir} - 
  ${getStatusColor(status)}${status}${chalk.reset()}`);
          } catch (e) {
            // Skip if no track.md
          }

        } else {

        }
      });
  }

  function getStatusColor(status: string): string {
    switch (status) {
      case 'PROPOSED': return chalk.blue;
      case 'ACTIVE': return chalk.yellow;
      case 'TESTING': return chalk.magenta;
      case 'COMPLETED': return chalk.green;
      case 'ARCHIVED': return chalk.gray;
      default: return chalk.white;
    }
  }

  function getStatusIcon(status: string): string {
    switch (status) {
      case 'PROPOSED': return '💡';
      case 'ACTIVE': return '🔨';
      case 'TESTING': return '🧪';
      case 'COMPLETED': return '✅';
      case 'ARCHIVED': return '📦';
      default: return '❓';
    }
  }
