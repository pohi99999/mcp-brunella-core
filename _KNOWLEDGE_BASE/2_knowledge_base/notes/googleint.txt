// package.json: "googleapis", "google-auth-library", "express", "nodemailer"
import express from "express";
import {GoogleAuth} from "google-auth-library";
import {google} from "googleapis";

const app = express();
app.use(express.json());

const WORKSPACE_SCOPES = [
  "https://www.googleapis.com/auth/gmail.send",
  "https://www.googleapis.com/auth/drive",
  "https://www.googleapis.com/auth/calendar",
  "https://www.googleapis.com/auth/documents",
  "https://www.googleapis.com/auth/spreadsheets.readonly"
];

async function asUser(serviceAccountEmail, userEmail){
  const auth = new GoogleAuth({
    scopes: WORKSPACE_SCOPES,
    clientOptions: { subject: userEmail }, // DWD impersonation
  });
  // Az ADC a futtató környezet SA-ját használja; DWD-hez az Admin Console-on kell beállítanod az SA-t és a scope-okat.
  const client = await auth.getClient();
  return { client };
}


1) POST /send-teaser-email – Gmail + Drive csatolmány

import MailComposer from "nodemailer/lib/mail-composer/index.js";

app.post("/send-teaser-email", async (req, res) => {
  const { sheetId, range, subject, bodyLangs, driveFileId, userEmail } = req.body;
  const { client } = await asUser(process.env.WORKSPACE_SA, userEmail);
  const sheets = google.sheets({version:"v4", auth: client});
  const drive  = google.drive({version:"v3", auth: client});
  const gmail  = google.gmail({version:"v1", auth: client});

  const rows = (await sheets.spreadsheets.values.get({ spreadsheetId: sheetId, range })).data.values || [];
  const fileMeta = await drive.files.get({ fileId: driveFileId, fields: "name,mimeType" });
  const fileData = await drive.files.get({ fileId: driveFileId, alt: "media" }, { responseType: "arraybuffer" });

  const tasks = rows.map(async ([email, lang="hu"]) => {
    const body = bodyLangs[lang] || bodyLangs.hu;
    const mail = new MailComposer({
      to: email,
      subject,
      text: body,
      attachments: [{
        filename: fileMeta.data.name,
        content: Buffer.from(fileData.data),
        contentType: fileMeta.data.mimeType
      }]
    });
    const raw = (await mail.compile().build()).toString("base64").replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
    await gmail.users.messages.send({ userId: "me", requestBody: { raw }});
  });

  await Promise.all(tasks);
  res.json({ ok: true, sent: rows.length });
});


2) POST /create-meet – Calendar esemény + Meet

app.post("/create-meet", async (req, res) => {
  const { title, startIso, durationMin, attendees=[], userEmail } = req.body;
  const { client } = await asUser(process.env.WORKSPACE_SA, userEmail);
  const calendar = google.calendar({version:"v3", auth: client});
  const start = new Date(startIso);
  const end = new Date(start.getTime() + durationMin*60000);

  const event = {
    summary: title,
    start: { dateTime: start.toISOString() },
    end:   { dateTime: end.toISOString() },
    attendees: attendees.map(e => ({ email: e })),
    conferenceData: { createRequest: { requestId: "req-"+Date.now() } }
  };

  const resp = await calendar.events.insert({
    calendarId: "primary",
    requestBody: event,
    conferenceDataVersion: 1
  });
  res.json({ ok: true, htmlLink: resp.data.htmlLink, meet: resp.data.hangoutLink });
});


3) POST /doc-from-template – Docs sablon klónozás + csere

app.post("/doc-from-template", async (req, res) => {
  const { templateId, name, folderId, placeholders={}, shareEmails=[], userEmail } = req.body;
  const { client } = await asUser(process.env.WORKSPACE_SA, userEmail);
  const drive = google.drive({version:"v3", auth: client});
  const docs  = google.docs({version:"v1", auth: client});

  // másolat
  const copy = await drive.files.copy({ fileId: templateId, requestBody: { name, parents: [folderId] }});
  const documentId = copy.data.id;

  // placeholder csere
  const requests = Object.entries(placeholders).map(([k,v]) => ({
    replaceAllText: { containsText: { text: `{{${k}}}`, matchCase: true }, replaceText: v }
  }));
  if (requests.length) await docs.documents.batchUpdate({ documentId, requestBody: { requests }});

  // megosztás
  await Promise.all(shareEmails.map(email => drive.permissions.create({
    fileId: documentId,
    requestBody: { type: "user", role: "reader", emailAddress: email },
    sendNotificationEmail: false
  })));

  res.json({ ok: true, documentId, link: `https://docs.google.com/document/d/${documentId}/edit` });
});

## cloudrun.yaml (opcionális infra-as-code)


```yaml
apiVersion: serving.knative.dev/v1
kind: Service
metadata:
  name: agentops
  namespace: default
spec:
  template:
    spec:
      serviceAccountName: workspace-sa@PROJECT_ID.iam.gserviceaccount.com
      containers:
        - image: gcr.io/PROJECT_ID/agentops:latest
          env:
            - name: DEFAULT_IMPERSONATE_USER
              value: you@your-domain.com
            - name: WORKSPACE_SA_KEY
              valueFrom:
                secretKeyRef:
                  name: WORKSPACE_SA_KEY
                  key: latest
      containerConcurrency: 80
```

## scripts/deploy.sh


```bash
#!/usr/bin/env bash
set -euo pipefail
PROJECT_ID=${PROJECT_ID:-"your-project"}
REGION=us-central1
SERVICE=agentops


# Secret (egyszeri)
#gcloud secrets create WORKSPACE_SA_KEY --replication-policy=automatic || true
#gcloud secrets versions add WORKSPACE_SA_KEY --data-file=sa-workspace.json


# Build & Deploy
gcloud run deploy "$SERVICE" \
  --source . \
  --region "$REGION" \
  --service-account=workspace-sa@${PROJECT_ID}.iam.gserviceaccount.com \
  --set-env-vars=DEFAULT_IMPERSONATE_USER="you@your-domain.com" \
  --update-secrets=WORKSPACE_SA_KEY=WORKSPACE_SA_KEY:latest \
  --no-allow-unauthenticated


RUN_URL=$(gcloud run services describe "$SERVICE" --region "$REGION" --format='value(status.url)')
echo "RUN_URL=$RUN_URL"


echo "Granting run.invoker to Extension SA..."
gcloud run services add-iam-policy-binding "$SERVICE" \
  --region "$REGION" \
  --member=serviceAccount:agent-ext@${PROJECT_ID}.iam.gserviceaccount.com \
  --role=roles/run.invoker


echo "All set."

Következő 5 lépés (gyors végrehajtás)

GCP projekt + régió

Projekt kiválasztása, régió: us-central1.

Két Service Account

agent-ext@… (Extension OIDC) – kulcs nélkül.

workspace-sa@… (DWD) – Domain-wide Delegation a Gmail/Drive/Calendar/Docs/Sheets scopokra.

A workspace-sa JSON kulcsát töltsd fel Secret Managerbe WORKSPACE_SA_KEY néven.

Cloud Run deploy

Futtasd a canvasban található scripts/deploy.sh lépéseit, vagy a README parancsait.

Adj roles/run.invoker jogot az agent-ext@… SA-nak a szolgáltatásra.

OpenAPI + Extension

Másold fel az openapi/spec.yaml-t GCS-be.

Hozd létre az Extensiont (OIDC→agent-ext@…) és add meg a spec URI-t.
