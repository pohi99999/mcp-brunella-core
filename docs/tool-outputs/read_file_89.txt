  const calculateProgress = (pipeline: PipelineData): number => {
    const total = pipeline.phases.length;
    const done = pipeline.phases.filter(
      (p) => p.status === "done" || p.status === "skipped",
    ).length;
    const running = pipeline.phases.filter(
      (p) => p.status === "running",
    ).length;
    return Math.round(((done + running * 0.5) / total) * 100);
